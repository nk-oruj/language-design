procedure TestingFibonachi
record
    index       gets [integer]
    value       sets [integer]

    elementA    [integer]
    elementB    [integer]
    elementC    [integer]

scheme

    load elementA with (0)
    load elementB with (1)

    scheme loop
        
        load elementC with (add elementA elementB)
        
        load elementA with (elementB)
        load elementB with (elementC)
        
        drop loop if (le index 0)
