MODULE LinkedList

PROMISE STRUCTURE LinkedListNode

STRUCTURE LinkedListNodeDesc
RECORD
    value : int64
    next : LinkedListNode
FINISH

STRUCTURE LinkedListNode AS POINT TO LinkedListNodeDesc

STRUCTURE LinkedListDesc
RECORD
    size : int64
    head : LinkedListNode
FINISH

STRUCTURE LinkedList AS POINT TO LinkedListDesc

PROCEDURE CreateNewNode(value : int64, next : LinkedListNode, OUT node : LinkedListNode)
SCHEME
    LOAD node WITH NEW POINT
    LOAD VALUE AT node.value WITH value
    LOAD VALUE AT node.next WITH node
FINISH

PROCEDURE PushFront(list : LinkedList, value : int64)
RECORD
    newNode : LinkedListNode
    size : int64
SCHEME
    CALL CreateNewNode WITH (value, VALUE AT list.head, OUT newNode)
    LOAD FIELD AT list.head WITH newNode
    
    LOAD size WITH VALUE AT list.size
    LOAD size WITH [add size 1]
    LOAD FIELD AT list.size WITH [size]
FINISH
